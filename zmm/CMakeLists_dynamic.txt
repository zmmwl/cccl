cmake_minimum_required(VERSION 3.12)
project(ZMM_Dynamic_Operations LANGUAGES CXX CUDA)

# 设置C++标准
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 设置CUDA标准
set(CMAKE_CUDA_STANDARD 14)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)

# 查找CUDA
find_package(CUDAToolkit REQUIRED)

# 包含CCCL路径
set(CCCL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
include_directories(${CCCL_ROOT})
include_directories(${CCCL_ROOT}/cub)
include_directories(${CCCL_ROOT}/thrust)
include_directories(${CCCL_ROOT}/libcudacxx/include)

# 设置CUDA编译选项
set(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} -lineinfo")
set(CMAKE_CUDA_FLAGS_DEBUG "${CMAKE_CUDA_FLAGS_DEBUG} -G -g")
set(CMAKE_CUDA_FLAGS_RELEASE "${CMAKE_CUDA_FLAGS_RELEASE} -O3")

# 检测GPU架构
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES "70;75;80;86")
endif()

# 设置输出目录
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

# 创建AddOperation动态库 (3列)
add_library(operations_add_3 SHARED operations_add.cu)
target_link_libraries(operations_add_3 CUDA::cudart)
set_target_properties(operations_add_3 PROPERTIES 
    CUDA_VISIBILITY_PRESET hidden
    CXX_VISIBILITY_PRESET hidden
)

# 创建AddOperation动态库 (5列)
add_library(operations_add_5 SHARED operations_add_5.cu)
target_link_libraries(operations_add_5 CUDA::cudart)
set_target_properties(operations_add_5 PROPERTIES 
    CUDA_VISIBILITY_PRESET hidden
    CXX_VISIBILITY_PRESET hidden
)

# 创建MultiplyOperation动态库 (3列)
add_library(operations_multiply_3 SHARED operations_multiply.cu)
target_link_libraries(operations_multiply_3 CUDA::cudart)
set_target_properties(operations_multiply_3 PROPERTIES 
    CUDA_VISIBILITY_PRESET hidden
    CXX_VISIBILITY_PRESET hidden
)

# 创建动态示例程序
add_executable(dynamic_example dynamic_example.cu)
target_link_libraries(dynamic_example CUDA::cudart dl)

# 创建原始示例程序
add_executable(example example.cu)
add_executable(simple_example simple_example.cu)
target_link_libraries(example CUDA::cudart)
target_link_libraries(simple_example CUDA::cudart)

# 设置编译器特定选项
if(CMAKE_CUDA_COMPILER_ID STREQUAL "NVIDIA")
    target_compile_options(operations_add_3 PRIVATE 
        $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda>
        $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>
        $<$<COMPILE_LANGUAGE:CUDA>:--compiler-options=-fPIC>
    )
    target_compile_options(operations_add_5 PRIVATE 
        $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda>
        $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>
        $<$<COMPILE_LANGUAGE:CUDA>:--compiler-options=-fPIC>
    )
    target_compile_options(operations_multiply_3 PRIVATE 
        $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda>
        $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>
        $<$<COMPILE_LANGUAGE:CUDA>:--compiler-options=-fPIC>
    )
    target_compile_options(dynamic_example PRIVATE 
        $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda>
        $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>
    )
    target_compile_options(example PRIVATE 
        $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda>
        $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>
    )
    target_compile_options(simple_example PRIVATE 
        $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda>
        $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>
    )
endif()

# 添加头文件到IDE
set(HEADER_FILES
    column_processor.cuh
    operations_interface.h
    dynamic_processor.h
)

# 创建一个仅包含头文件的目标，方便IDE显示
add_custom_target(headers SOURCES ${HEADER_FILES})

# 显示一些有用的信息
message(STATUS "CUDA Compiler: ${CMAKE_CUDA_COMPILER}")
message(STATUS "CUDA Version: ${CUDAToolkit_VERSION}")
message(STATUS "CUDA Architectures: ${CMAKE_CUDA_ARCHITECTURES}")
message(STATUS "Build Type: ${CMAKE_BUILD_TYPE}")

# 添加测试目标
enable_testing()
add_test(NAME simple_test COMMAND simple_example)
add_test(NAME full_test COMMAND example)
add_test(NAME dynamic_test COMMAND dynamic_example)

# 自定义目标：构建所有
add_custom_target(build_all 
    DEPENDS operations_add_3 operations_add_5 operations_multiply_3 dynamic_example example simple_example
)

# 帮助信息
add_custom_target(show_help
    COMMAND ${CMAKE_COMMAND} -E echo "可用目标："
    COMMAND ${CMAKE_COMMAND} -E echo "  operations_add_3     - AddOperation动态库(3列)"
    COMMAND ${CMAKE_COMMAND} -E echo "  operations_add_5     - AddOperation动态库(5列)"
    COMMAND ${CMAKE_COMMAND} -E echo "  operations_multiply_3 - MultiplyOperation动态库(3列)"
    COMMAND ${CMAKE_COMMAND} -E echo "  dynamic_example      - 动态加载示例"
    COMMAND ${CMAKE_COMMAND} -E echo "  example              - 完整性能测试示例"
    COMMAND ${CMAKE_COMMAND} -E echo "  simple_example       - 简单使用示例"
    COMMAND ${CMAKE_COMMAND} -E echo "  build_all            - 构建所有目标"
)
